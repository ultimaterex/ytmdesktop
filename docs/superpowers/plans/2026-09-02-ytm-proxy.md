# YTM Desktop Proxy Support Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a settings-driven HTTP/SOCKS proxy that applies only to the YouTube Music `BrowserView` session so personal geo checks can use a chosen egress.

**Architecture:** Persist proxy config in the existing `conf` store; apply via `session.fromPartition(...).setProxy()` on `persist:ytmview` / `persist:ytmview-dev` before Music loads; handle proxy auth on `app.on('login')`; harden WebRTC when enabled; expose a Network tab in Settings.

**Tech Stack:** Electron 40, Vue 3, TypeScript, `conf`, Electron Forge / Yarn 4

**Spec:** `docs/superpowers/specs/2026-09-02-ytm-proxy-design.md`

```js
/** @typedef {"http"|"https"|"socks4"|"socks5"} ProxyProtocol */

/**
 * @param {{ enabled: boolean, protocol: ProxyProtocol, host: string, port: number, bypassRules: string }} input
 */
export function isProxyConfigured(input) {
  const host = (input.host ?? "").trim();
  const port = Number(input.port);
  return Boolean(input.enabled && host && Number.isInteger(port) && port > 0 && port <= 65535);
}

/**
 * @param {{ enabled: boolean, protocol: ProxyProtocol, host: string, port: number, bypassRules: string }} input
 * @returns {{ mode: string, proxyRules?: string, proxyBypassRules?: string }}
 */
export function buildElectronProxyConfig(input) {
  if (!isProxyConfigured(input)) {
    return { mode: "direct" };
  }
  const host = input.host.trim();
  const protocol = input.protocol;
  return {
    mode: "fixed_servers",
    proxyRules: `${protocol}://${host}:${Number(input.port)}`,
    proxyBypassRules: (input.bypassRules ?? "").trim() || "<local>"
  };
}
```

- [ ] **Step 1: Implement `apply-ytm-proxy.ts`**

```ts
import { app, session, Session } from "electron";
import Conf from "conf";
import log from "electron-log";
import { safeStorage } from "electron";
import { StoreSchema } from "../../shared/store/schema";
import { buildElectronProxyConfig } from "./build-proxy-config";

export function getYtmPartitionName(isPackaged: boolean): string {
  return isPackaged ? "persist:ytmview" : "persist:ytmview-dev";
}

export function getYtmSession(isPackaged: boolean): Session {
  return session.fromPartition(getYtmPartitionName(isPackaged));
}

function decryptPassword(passwordEncrypted: string | null): string {
  if (!passwordEncrypted) return "";
  if (!safeStorage.isEncryptionAvailable()) return "";
  try {
    return safeStorage.decryptString(Buffer.from(passwordEncrypted, "base64"));
    // NOTE: confirm how this app encodes safeStorage payloads.
    // Companion tokens use IPC encryptString which returns a string already —
    // match that exact encode/decode (see existing safeStorage:encryptString handlers).
  } catch (e) {
    log.error("Failed to decrypt proxy password", e);
    return "";
  }
}

export async function applyYtmProxyFromStore(store: Conf<StoreSchema>): Promise<void> {
  const proxy = store.get("proxy");
  const config = buildElectronProxyConfig(proxy);
  const ses = getYtmSession(app.isPackaged);
  await ses.setProxy(config);
  log.info(`YTM proxy mode=${config.mode}${config.proxyRules ? ` rules=${config.proxyRules}` : ""}`);
}

export function resolveProxyCredentials(store: Conf<StoreSchema>): { username: string; password: string } | null {
  const proxy = store.get("proxy");
  if (!buildElectronProxyConfig(proxy).proxyRules) return null;
  const username = (proxy.username ?? "").trim();
  const password = decryptPassword(proxy.passwordEncrypted);
  if (!username && !password) return null;
  return { username, password };
}

let proxyAuthAttached = false;
export function attachProxyAuthHandler(store: Conf<StoreSchema>): void {
  if (proxyAuthAttached) return;
  proxyAuthAttached = true;
  app.on("login", (event, _webContents, _details, authInfo, callback) => {
    if (!authInfo.isProxy) {
      return;
    }
    const creds = resolveProxyCredentials(store);
    if (!creds) {
      return;
    }
    event.preventDefault();
    callback(creds.username, creds.password);
  });
}
```

**Critical:** Open `src/main/index.ts` `safeStorage:encryptString` / `decryptString` handlers and **match their string format exactly** (they likely pass through Electron’s string API, not raw base64 Buffers). Adjust `decryptPassword` accordingly — do not invent a new encoding.
